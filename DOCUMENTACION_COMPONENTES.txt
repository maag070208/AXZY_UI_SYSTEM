# AXZY UI - TECHNICAL SPECIFICATIONS & USAGE

Documentación detallada de componentes con tipado estricto y ejemplos de implementación.
Todos los componentes consumen tokens de `theme.ts`.

---

## ITLayout
Contenedor raíz del ecosistema administrativo.

### Model
```ts
export interface ITLayoutProps {
  topBar: ITTopBarProps;      // Configuración de Navbar superior
  sidebar: ITSidebarProps;    // Configuración de Sidebar lateral
  children: React.ReactNode;  // Contenido de la pantalla
  className?: string;         // Clases para el wrapper principal
  contentClassName?: string;  // Clases para el área de contenido (main)
}
```

### Usage
```tsx
<ITLayout topBar={topBarProps} sidebar={sidebarProps}>
  <DashboardScreen />
</ITLayout>
```

## ITSearchTable
Tabla "todo-en-uno" avanzada con edición en línea y búsqueda global.

### Model
```ts
export interface SearchColumn<T = any> {
  key: string;
  label: string;
  type: "string" | "number" | "boolean" | "date" | "actions" | "catalog";
  filter?: boolean | "catalog"; 
  sortable?: boolean;
  editable?: boolean;
  inputType?: "text" | "number" | "select" | "checkbox" | "date";
  options?: { value: string | number; label: string }[];
  render?: (row: T) => React.ReactNode;
}

export interface ITSearchTableProps<T> {
  columns: SearchColumn<T>[];
  data: T[];
  title?: string;
  pageIndex: number;
  totalCount: number;
  totalPages: number;
  onPageChange?: (page: number) => void;
  onSortChange?: (config: { key: string, direction: "asc" | "desc" }) => void;
  onFilterChange?: (filters: Record<string, any>) => void;
  variant?: "default" | "striped" | "bordered" | "clean";
  size?: "sm" | "md" | "lg";
}
```

### Usage
```tsx
<ITSearchTable
  title="Gestión de Inventario"
  columns={inventoryColumns}
  data={products}
  totalCount={100}
  totalPages={10}
  pageIndex={1}
  onPageChange={(p) => setPage(p)}
/>
```

## ITFormBuilder
Motor dinámico de formularios basado en configuración JSON.

### Model
```ts
export interface FieldConfig {
  name: string;
  label: string;
  type: "text" | "email" | "password" | "number" | "select" | "date" | "checkbox";
  required?: boolean;
  options?: { label: string; value: any }[];
}

export interface ITFormBuilderProps {
  config: FieldConfig[];
  values: any;
  handleChange: (e: any) => void;
  errors?: Record<string, string>;
  touched?: Record<string, boolean>;
}
```

### Usage
```tsx
<ITFormBuilder
  values={formik.values}
  handleChange={formik.handleChange}
  config={[
    { name: 'email', label: 'Email', type: 'email', required: true },
    { name: 'role', label: 'Rol', type: 'select', options: [{label: 'Admin', value: '1'}]}
  ]}
/>
```
